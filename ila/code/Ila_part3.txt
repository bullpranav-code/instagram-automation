function adminHtml_() {
  const ss = ss_(), sh = n => ss.getSheetByName(TABS[n][0]), esc = anEsc_;
  const d = x => x instanceof Date ? Utilities.formatDate(x, 'Asia/Kolkata', 'dd MMM HH:mm') : esc(x);
  const leads = anAll_(sh('leads'), 7), chats = anAll_(sh('chats'), 7), reps = anAll_(sh('reports'), 7), fbs = anAll_(sh('feedback'), 7), svcs = anAll_(sh('services'), 9);
  const days = anDays_(reps, 0, 30), cdays = anDays_(chats, 0, 30);
  const scores = reps.map(r => Number(r[5])).filter(x => x > 0);
  const avg = scores.length ? (scores.reduce((a, b) => a + b, 0) / scores.length).toFixed(1) : '-';
  const ready = scores.length ? Math.round(scores.filter(x => x >= 7).length / scores.length * 100) + '%' : '-';
  const up = fbs.filter(r => /👍/.test(r[4])).length, rated = fbs.filter(r => /👍|👎/.test(r[4])).length;
  const wk = days.slice(-7).reduce((a, x) => a + x.rows.length, 0), pwk = days.slice(-14, -7).reduce((a, x) => a + x.rows.length, 0);
  const flagged = chats.slice(-300).filter(r => /handoff|crisis/.test(r[6])).length;
  const modes = {}; chats.forEach(r => { const m = r[3] || 'chat'; modes[m] = (modes[m] || 0) + 1; });
  const roles = {}; reps.forEach(r => { if (r[3]) roles[r[3]] = (roles[r[3]] || 0) + 1; });
  const dist = {}; ['0–4 (needs work)', '5–6 (getting there)', '7–8 (ready)', '9–10 (strong)'].forEach(k => dist[k] = 0);
  scores.forEach(x => { dist[x < 5 ? '0–4 (needs work)' : x < 7 ? '5–6 (getting there)' : x < 9 ? '7–8 (ready)' : '9–10 (strong)']++; });
  const weak = { 'Salary answer': 0, 'Using numbers / proof': 0, 'Closing / next step': 0, 'Grammar & words': 0, 'Confidence': 0, 'Structure (STAR)': 0 };
  reps.forEach(r => { const t = (String(r[6]).split(/CORRECTIONS:/i)[1] || '').toLowerCase();
    if (/salary|ctc|expect/.test(t)) weak['Salary answer']++; if (/number|figure|data|result/.test(t)) weak['Using numbers / proof']++;
    if (/close|closing|next step|callback/.test(t)) weak['Closing / next step']++; if (/grammar|words:|tense/.test(t) && !/words:\s*none/.test(t)) weak['Grammar & words']++;
    if (/confiden|hesitat|filler/.test(t)) weak['Confidence']++; if (/star|structure|situation/.test(t)) weak['Structure (STAR)']++; });
  const table = (head, rows, empty) => '<div style="overflow-x:auto"><table><tr>' + head.map(h => '<th>' + h + '</th>').join('') + '</tr>' + (rows || '<tr><td colspan=' + head.length + '>' + empty + '</td></tr>') + '</table></div>';
  const lastRep = reps.slice(-15).reverse().map(r => { const sc = Number(r[5]) || 0, col = sc >= 7 ? '#16A34A' : sc >= 5 ? '#C97A2B' : '#B42318';
    return '<tr><td>' + d(r[0]) + '</td><td>' + esc(r[2]) + '</td><td>' + esc(r[3]) + '</td><td>' + esc(r[4]) + '</td><td><b style="color:' + col + '">' + esc(r[5]) + '</b></td><td>' + esc((String(r[6]).match(/READY:\s*([^\n]+)/i) || [])[1] || '') + '</td></tr>'; }).join('');
  const lastFb = fbs.slice(-15).reverse().map(r => '<tr><td>' + d(r[0]) + '</td><td>' + esc(r[1]) + '</td><td>' + esc(r[3]) + '</td><td>' + esc(r[4]) + '</td><td>' + esc(r[5]) + '</td></tr>').join('');
  const lastLeads = leads.slice(-15).reverse().map(r => '<tr><td>' + d(r[0]) + '</td><td>' + esc(r[2]) + '</td><td>' + esc(r[3]) + '</td><td>' + esc(r[5]) + '</td><td>' + esc(r[6]) + '</td></tr>').join('');
  const lastSvc = svcs.slice(-15).reverse().map(r => '<tr><td>' + d(r[0]) + '</td><td>' + esc(r[2]) + '</td><td>' + esc(r[3]) + '</td><td>' + esc(r[4]) + '</td><td>' + esc(r[7]) + '</td><td>' + (r[8] ? '<a href="' + esc(r[8]) + '" target="_blank">Open</a>' : '') + '</td></tr>').join('');
  const key = encodeURIComponent(PropertiesService.getScriptProperties().getProperty('ADMIN_KEY') || '');
  return '<link href="https://fonts.googleapis.com/css2?family=Poppins:wght@400;600;700&display=swap" rel="stylesheet">' +
    '<style>body{margin:0;background:#F7FAF8;font-family:Poppins,Arial;color:#0B3D1E}table{border-collapse:collapse;width:100%;background:#fff;font-size:13px}th{background:#0B3D1E;color:#fff;text-align:left;font-weight:600}th,td{padding:8px 10px;border-bottom:1px solid #DCE5DE;vertical-align:top}a{color:#16A34A}.two{display:grid;grid-template-columns:1fr 1fr;gap:14px}@media(max-width:760px){.two{grid-template-columns:1fr}}</style>' +
    '<div style="background:#0B3D1E;padding:16px 20px;display:flex;justify-content:space-between;align-items:center"><div><div style="color:#79C24A;font-size:11px;letter-spacing:3px;font-weight:600">ACCENT · ILA · BETA</div><div style="color:#fff;font-size:20px;font-weight:600">Analytics dashboard</div></div><img src="data:image/png;base64,' + LOGO_LIGHT + '" style="height:34px"></div>' +
    '<div style="max-width:1100px;margin:0 auto;padding:12px 20px 40px">' +
    '<p style="font-size:13px;color:#8A968E">Updated ' + Utilities.formatDate(new Date(), 'Asia/Kolkata', 'dd MMM yyyy, HH:mm') + ' · every record is kept in the sheet · <a href="' + ss.getUrl() + '" target="_blank">Open sheet</a> · <a href="' + CONFIG.KILI_URL + '?view=calls&key=' + key + '" target="_top">Kili dashboard →</a></p>' +
    (flagged ? '<p style="background:#FDECEA;color:#B42318;padding:10px 14px;border-radius:12px"><b>' + flagged + ' chats need a human.</b> Check rows flagged handoff/crisis in "Ila Chats".</p>' : '') +
    anKpi_([['STUDENTS', leads.length, 'signed up'], ['CHAT MESSAGES', chats.length, ''], ['MOCK INTERVIEWS', reps.length, wk + ' this week (' + (wk >= pwk ? '+' : '') + (wk - pwk) + ')'],
      ['AVG SCORE', avg + (avg === '-' ? '' : '/10'), 'all interviews'], ['INTERVIEW-READY', ready, 'scored 7+'], ['BETA 👍', rated ? Math.round(up / rated * 100) + '%' : '-', rated + ' ratings']]) +
    anCard_('Mock interviews per day · average score (last 30 days)', anChart_(days.map(x => x.label), days.map(x => x.rows.length),
      days.map(x => { const v = x.rows.map(r => Number(r[5])).filter(y => y > 0); return v.length ? v.reduce((a, b) => a + b, 0) / v.length : null; }), 10, 'Interviews', 'Avg score /10')) +
    anCard_('Ila chat messages per day (last 30 days)', anChart_(cdays.map(x => x.label), cdays.map(x => x.rows.length), null, 0, 'Messages', '')) +
    '<div class="two">' + anCard_('Score spread', anHBars_(dist, 4)) + anCard_('Most common weak areas (from reports)', anHBars_(weak, 6)) + '</div>' +
    '<div class="two">' + anCard_('Roles practised', anHBars_(roles, 6)) + anCard_('What students use Ila for', anHBars_(modes, 7)) + '</div>' +
    anCard_('Latest mock interviews', table(['When', 'Name', 'Role', 'Lang', 'Score', 'Ready?'], lastRep, 'None yet')) +
    anCard_('Beta feedback', table(['When', 'Product', 'Name', 'Rating', 'Comment'], lastFb, 'None yet')) +
    '<div class="two">' + anCard_('Latest students', table(['When', 'Name', 'WhatsApp', 'I am', 'District'], lastLeads, 'None yet')) +
    anCard_('Service requests', table(['When', 'Name', 'WhatsApp', 'Service', 'Status', 'Draft'], lastSvc, 'None yet')) + '</div></div>';
}

/* ================= helpers ================= */
function ss_() {
  const id = String(PropertiesService.getScriptProperties().getProperty('SHEET_ID') || CONFIG.SHEET_ID || '').trim();
  return id ? SpreadsheetApp.openById(id) : SpreadsheetApp.getActiveSpreadsheet();
}
function tab_(ss, k) {
  const [name, headers] = TABS[k];
  let s = ss.getSheetByName(name);
  if (!s) {
    s = ss.insertSheet(name);
    s.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold').setBackground('#0B3D1E').setFontColor('#FFFFFF');
    s.setFrozenRows(1);
  } else if (s.getLastColumn() < headers.length) {
    s.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold').setBackground('#0B3D1E').setFontColor('#FFFFFF');
  }
  return s;
}
function overLimit_(session) {
  const c = CacheService.getScriptCache(), key = 'n_' + session + '_' + Utilities.formatDate(new Date(), 'Asia/Kolkata', 'yyyyMMdd');
  const n = Number(c.get(key) || 0) + 1;
  c.put(key, String(n), 21600);
  return n > CONFIG.DAILY_MESSAGE_LIMIT;
}
