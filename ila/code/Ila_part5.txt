    g += '<rect x="' + x + '" y="' + (H - P - h) + '" width="' + (bw * .64) + '" height="' + h + '" rx="3" fill="#16A34A"><title>' + anEsc_(labels[i]) + ': ' + v + '</title></rect>'; });
  if (line) { var pts = []; line.forEach(function (v, i) { if (v == null) return; pts.push((P + i * bw + bw / 2) + ',' + (P + (H - P * 2) * (1 - v / lineMax))); });
    if (pts.length) g += '<polyline points="' + pts.join(' ') + '" fill="none" stroke="#0B3D1E" stroke-width="2.5"/>' + pts.map(function (p) { var a = p.split(','); return '<circle cx="' + a[0] + '" cy="' + a[1] + '" r="3.5" fill="#0B3D1E"/>'; }).join(''); }
  var step = Math.ceil(n / 8);
  labels.forEach(function (l, i) { if (i % step === 0 || (i === n - 1 && (n - 1) % step > step / 2)) g += '<text x="' + (P + i * bw + bw / 2) + '" y="' + (H - P + 16) + '" font-size="10" text-anchor="middle" fill="#8A968E">' + anEsc_(l) + '</text>'; });
  g += '<rect x="' + P + '" y="' + (H + 12) + '" width="10" height="10" fill="#16A34A"/><text x="' + (P + 14) + '" y="' + (H + 21) + '" font-size="11" fill="#2C3A31">' + anEsc_(barName) + '</text>';
  if (line) g += '<line x1="' + (P + 150) + '" x2="' + (P + 164) + '" y1="' + (H + 17) + '" y2="' + (H + 17) + '" stroke="#0B3D1E" stroke-width="2.5"/><text x="' + (P + 168) + '" y="' + (H + 21) + '" font-size="11" fill="#2C3A31">' + anEsc_(lineName) + '</text>';
  return g + '</svg>';
}
/** Horizontal bars for a {label: count} map. */
function anHBars_(map, top) {
  var arr = Object.keys(map).map(function (k) { return [k, map[k]]; }).sort(function (a, b) { return b[1] - a[1]; }).slice(0, top || 6);
  var max = arr.length ? arr[0][1] : 1;
  return arr.length ? arr.map(function (a) { return '<div style="display:flex;align-items:center;gap:10px;margin:6px 0;font-size:13px"><span style="width:44%;color:#0B3D1E">' + anEsc_(a[0]) +
    '</span><div style="flex:1;height:10px;background:#DCE5DE;border-radius:6px;overflow:hidden"><i style="display:block;height:100%;width:' + Math.round(a[1] / max * 100) + '%;background:#16A34A"></i></div><b style="width:30px;text-align:right">' + a[1] + '</b></div>'; }).join('')
    : '<p style="color:#8A968E;font-size:13px">No data yet.</p>';
}
