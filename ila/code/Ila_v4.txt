/** API for the interview page hosted outside Google (it posts JSON here).
 *  action 'chat' (interview turns), 'tts' (natural Gemini voice), 'stt' (voice answer -> text). */
function doPost(e) {
  let out;
  try {
    const req = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    if (req.action === 'admin') out = adminData_(req.key);
    else if (req.action === 'community_links') out = communityLinks_();
    else if (req.action === 'admin_links') out = adminLinks_(req);
    else if (req.action === 'admin_posts') out = adminPosts_(req);
    else if (req.action === 'admin_approve') out = adminApprove_(req);
    else if (req.action === 'start') out = startSession(req);
    else if (!/^IL-[A-Z0-9]{8}$/.test(String(req.session || ''))) throw new Error('Session expired. Please sign in again.');
    else if (req.action === 'service') out = requestService(req);
    else if (req.action === 'course_state') out = courseState_(req);
    else if (req.action === 'course_join') out = courseJoin_(req);
    else if (req.action === 'course_event') out = courseEvent_(req);
    else if (req.action === 'green_submit') out = greenSubmit_(req);
    else if (req.action === 'course_video') out = courseVideo_(req);
    else if (req.action === 'feedback') out = saveFeedback_(req);
    else if (req.action === 'tts') { if (overCap_(req.session, 'tts', 40)) throw new Error('Voice limit reached for today'); out = { audio: ilaTts_(String(req.text || '').slice(0, 600), req.lang) }; }
    else if (req.action === 'stt') { if (overCap_(req.session, 'stt', 40)) throw new Error('Recording limit reached for today. Please type your answer.'); out = { text: ilaStt_(req) }; }
    else out = chat(req);   // career / resume / linkedin / mentor / interview turns
  } catch (err) { out = { error: String(err.message || err) }; }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

/** Natural voice (Gemini TTS). Returns base64 WAV. Throws if Google is busy, so the page falls back to text. */
function ilaTts_(text, lang) {
  if (!text) throw new Error('No text');
  const key = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  const style = lang === 'ml-IN'
    ? 'Speak in clear, standard Malayalam like a calm, professional HR manager. Warm, unhurried, confident.'
    : 'Speak in clear Indian English like a calm, professional HR manager. Warm, unhurried, confident.';
  const models = ['gemini-3.8-flash-tts', 'gemini-3.8-flash-lite-tts', 'gemini-3.1-flash-tts-preview'];
  for (let m = 0; m < models.length; m++) {
    const r = UrlFetchApp.fetch('https://generativelanguage.googleapis.com/v1beta/interactions', {
      method: 'post', contentType: 'application/json', muteHttpExceptions: true, headers: { 'x-goog-api-key': key },
      payload: JSON.stringify({ model: models[m],
        input: [{ type: 'user_input', content: [{ type: 'text', text: text, annotations: [{ type: 'speech_metadata', style: style }] }] }],
        response_format: { type: 'audio', mime_type: 'audio/wav' },
        generation_config: { speech_config: [{ voice: 'Puck' }] } }) });
    if (r.getResponseCode() !== 200) continue;
    const json = JSON.parse(r.getContentText());
    let b64 = '';
    (json.steps || []).forEach(st => (st.content || []).forEach(c => { if (c.type === 'audio' && c.data) b64 = c.data; }));
    if (!b64 && json.output_audio) b64 = json.output_audio.data || '';
    if (!b64) continue;
    const bytes = Utilities.base64Decode(b64);
    const head = String.fromCharCode(bytes[0] & 255, bytes[1] & 255, bytes[2] & 255, bytes[3] & 255);
    return head === 'RIFF' ? b64 : Utilities.base64Encode(wav_(bytes, 24000));
  }
  throw new Error('Voice is busy');
}
function overCap_(session, kind, max) {
  const c = CacheService.getScriptCache(), k = kind + '_' + session + '_' + Utilities.formatDate(new Date(), 'Asia/Kolkata', 'yyyyMMdd');
  const n = Number(c.get(k) || 0) + 1; c.put(k, String(n), 21600); return n > max;
}

function wav_(pcm, rate) {
  const h = [], s = t => { for (let i = 0; i < t.length; i++) h.push(t.charCodeAt(i)); };
  const u32 = v => h.push(v & 255, (v >> 8) & 255, (v >> 16) & 255, (v >> 24) & 255), u16 = v => h.push(v & 255, (v >> 8) & 255);
  s('RIFF'); u32(36 + pcm.length); s('WAVE'); s('fmt '); u32(16); u16(1); u16(1); u32(rate); u32(rate * 2); u16(2); u16(16); s('data'); u32(pcm.length);
  return h.map(b => b > 127 ? b - 256 : b).concat(pcm);
}

/** Voice answer -> text (Gemini listens to the recording). */
function ilaStt_(req) {
  const data = String(req.audio || '');
  if (!data) throw new Error('No recording');
  if (data.length > 6 * 1024 * 1024) throw new Error('Recording too long. Keep answers under 2 minutes.');
  const mime = /mp4|m4a|aac/.test(req.mime || '') ? 'audio/aac' : /ogg/.test(req.mime || '') ? 'audio/ogg' : /wav/.test(req.mime || '') ? 'audio/wav' : /webm/.test(req.mime || '') ? 'audio/webm' : 'audio/aac';
  const ask = 'Transcribe this spoken interview answer exactly as said, in ' + (req.lang === 'ml-IN' ? 'Malayalam script (keep English words in English)' : 'English') +
    '. Fix nothing, add nothing, no labels or notes. If there is no speech, reply exactly: (no speech)';
  const out = gemini_('You are a precise transcriber.', [{ role: 'user', parts: [{ text: ask }, { inlineData: { mimeType: mime, data: data } }] }],
    { temperature: 0, maxOutputTokens: 800, thinkingConfig: { thinkingBudget: 0 } });
  if (BUSY_RE.test(out)) throw new Error('Google is busy. Please record again in a moment.');
  return String(out).trim();
}

/* ================= course portal ================= */
function waKey_(w) { const d = String(w || '').replace(/\D/g, ''); return d.slice(-10); }
function enrollRow_(wa) {
  const sh = tab_(ss_(), 'enroll'), k = waKey_(wa);
  if (!k || sh.getLastRow() < 2) return { sh: sh, row: 0, v: null };
  const vals = sh.getRange(2, 1, sh.getLastRow() - 1, 14).getValues();
  for (let i = vals.length - 1; i >= 0; i--) if (waKey_(vals[i][2]) === k) return { sh: sh, row: i + 2, v: vals[i] };
  return { sh: sh, row: 0, v: null };
}
function pinHash_(wa, pin) {
  return Utilities.base64Encode(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, 'accent|' + waKey_(wa) + '|' + String(pin || ''), Utilities.Charset.UTF_8));
}
/** Throws unless the PIN matches (only when the learner has set one). */
function checkPin_(e, r) {
  if (!e.v || !e.v[13]) return;
  const c = CacheService.getScriptCache(), k = 'pinfail_' + waKey_(r.wa), fails = Number(c.get(k) || 0);
  if (fails >= 5) throw new Error('PIN_LOCKED');
  if (!r.pin) throw new Error('PIN_REQUIRED');
  if (pinHash_(r.wa, r.pin) !== e.v[13]) { c.put(k, String(fails + 1), 3600); throw new Error('PIN_WRONG'); }
}
function courseState_(r) {
  const e = enrollRow_(r.wa); checkPin_(e, r);
  const status = e.v ? String(e.v[4]) : 'none', approved = status === 'Approved';
  const k = waKey_(r.wa), done = {}, quiz = {};
  const ps = ss_().getSheetByName(TABS.progress[0]);
  if (ps && ps.getLastRow() > 1 && k) ps.getRange(2, 1, ps.getLastRow() - 1, 6).getValues().forEach(x => {
    if (waKey_(x[1]) !== k) return; if (x[4] === 'watched') done[x[3]] = 1; if (x[4] === 'check') quiz[x[3]] = Number(x[5]) || 0; });
  const videos = {};
  CONFIG.FREE_LESSONS.forEach(n => { videos[n] = CONFIG.FREE_VIDEO_BASE + n + '.mp4'; });
  if (approved) for (let n = 1; n <= 10; n++) if (!videos[n]) videos[n] = CONFIG.HOSTED_LESSONS.indexOf(n) > -1 ? CONFIG.FREE_VIDEO_BASE + (n < 10 ? n : '') + '.mp4' : 'private';
  return { status: status, videos: videos, done: done, checks: quiz, green: e.v ? String(e.v[8] || '') : '', points: e.v ? Number(e.v[9]) || 0 : 0,
    cert: e.v ? String(e.v[10] || '') : '', final: e.v ? String(e.v[11] || '') : '' };
}
/** Private lesson video from Drive, only for approved learners. File names must contain L03, L04 … L10. */
function courseVideo_(r) {
  const n = Number(r.lesson) || 0, e = enrollRow_(r.wa); checkPin_(e, r);
  if (!(e.v && e.v[4] === 'Approved')) throw new Error('This lesson unlocks after approval.');
  const tag = 'L' + (n < 10 ? '0' + n : n), it = DriveApp.getFolderById(CONFIG.VIDEO_FOLDER_ID).getFiles();
  let best = null;
  while (it.hasNext()) { const f = it.next(), nm = f.getName(); if (nm.indexOf(tag) > -1 && /\.(mp4|m4v|mov)$/i.test(nm)) { if (!best || f.getLastUpdated() > best.getLastUpdated()) best = f; } }
  if (!best) throw new Error('Lesson ' + n + ' video is being uploaded. Please check again soon.');
  if (best.getSize() > 25 * 1048576) throw new Error('Lesson video is too large to stream. Ask Accent to compress it.');
  return { mime: best.getMimeType() || 'video/mp4', data: Utilities.base64Encode(best.getBlob().getBytes()) };
}
function courseJoin_(r) {
  const e = enrollRow_(r.wa);
  if (e.v && /Pending|Approved/.test(e.v[4])) { if (!e.v[13] && /^\d{4}$/.test(String(r.pin))) e.sh.getRange(e.row, 14).setValue(pinHash_(r.wa, r.pin)); return courseState_(r); }
  if (!/^\d{4}$/.test(String(r.pin || ''))) throw new Error('Choose a 4-digit PIN.');
  tab_(ss_(), 'enroll').appendRow([new Date(), String(r.name || '').slice(0, 80), String(r.wa || ''), 'Sales Basics', 'Pending', String(r.note || '').slice(0, 300), '', '', '', 0, '', '', r.session || '', pinHash_(r.wa, r.pin)]);
  return courseState_(r);
}
function courseEvent_(r) {
  const n = Number(r.lesson) || 0, ev = String(r.event || '');
  if (!/^(watched|check|final)$/.test(ev)) throw new Error('Unknown event');
  const e = enrollRow_(r.wa); checkPin_(e, r);
  if (n > 2 && ev !== 'final' && !(e.v && e.v[4] === 'Approved')) throw new Error('This lesson unlocks after approval.');
  tab_(ss_(), 'progress').appendRow([new Date(), String(r.wa || ''), String(r.name || ''), n, ev, Number(r.score) || 0]);
  if (ev === 'final' && e.row) {
    const sc = Math.max(0, Math.min(10, Number(r.score) || 0));
    e.sh.getRange(e.row, 12).setValue(sc + '/10');
    if (sc >= 7 && !e.v[10]) {
      const id = 'ACC-SB-' + Utilities.formatDate(new Date(), 'Asia/Kolkata', 'yyMMdd') + '-' + Utilities.getUuid().slice(0, 4).toUpperCase();
      e.sh.getRange(e.row, 10, 1, 2).setValues([[(Number(e.v[9]) || 0) + 200, id]]);   // Leaf Points: finish a short course = 200
    }
  }
  return courseState_(r);
}
function greenSubmit_(r) {
  const e = enrollRow_(r.wa); checkPin_(e, r);
  if (!e.row) throw new Error('Join the course first.');
  const bytes = Utilities.base64Decode(String(r.photo || ''));
  if (!bytes.length || bytes.length > 6 * 1048576) throw new Error('Send a photo under 6 MB.');
  const url = folder_('Green Task').createFile(Utilities.newBlob(bytes, 'image/jpeg', waKey_(r.wa) + '_' + Utilities.formatDate(new Date(), 'Asia/Kolkata', 'yyyyMMdd_HHmm') + '.jpg')).getUrl();
  e.sh.getRange(e.row, 8, 1, 2).setValues([[url, 'Submitted' + (r.insta ? ' · ' + String(r.insta).slice(0, 120) : '')]]);
  return courseState_(r);
}
function adminApprove_(r) {
  const k = PropertiesService.getScriptProperties().getProperty('ADMIN_KEY');
  if (!k || r.key !== k) throw new Error('Wrong admin key.');
  const e = enrollRow_(r.wa);
  if (!e.row) throw new Error('Learner not found.');
  if (r.what === 'access') e.sh.getRange(e.row, 5, 1, 3).setValues([['Approved', e.v[5], new Date()]]);
  else if (r.what === 'green') e.sh.getRange(e.row, 9, 1, 2).setValues([['Approved · refund due', (Number(e.v[9]) || 0) + 50]]);   // Green Task planting post = 50 LP
  else if (r.what === 'refunded') e.sh.getRange(e.row, 9).setValue('Refunded');
  else if (r.what === 'resetpin') { e.sh.getRange(e.row, 14).setValue(''); CacheService.getScriptCache().remove('pinfail_' + waKey_(r.wa)); }
  return { ok: true };
}

/** One chat turn. history = [{role:'user'|'model', text}] (last few turns, kept in the browser). */
function chat(req) {
  const session = String(req.session || ''), name = String(req.name || '');
  if (!/^IL-[A-Z0-9]{8}$/.test(session)) throw new Error('Session expired. Refresh the page.');
  if (overLimit_(session)) return { reply: "You've reached today's chat limit. Ila will be back tomorrow, or WhatsApp us on +91 99958 42727.", flag: 'limit' };

  const mode = req.mode || 'chat';
  const msg = String(req.message || '').slice(0, 4000);
  let fileUrl = '';
  const contents = (req.history || []).slice(mode === 'interview' ? -14 : -6).map(h => ({ role: h.role === 'model' ? 'model' : 'user', parts: [{ text: String(h.text).slice(0, 4000) }] }));
  const parts = [{ text: (mode !== 'chat' ? '[MODE: ' + mode + ']\n' : '') + (msg || '(attached file)') }];
  if (req.file && req.file.data) {
    const bytes = Utilities.base64Decode(req.file.data);
    if (bytes.length > CONFIG.MAX_UPLOAD_MB * 1048576) throw new Error('File is over ' + CONFIG.MAX_UPLOAD_MB + ' MB.');
    const mime = /pdf$/i.test(req.file.name || '') ? 'application/pdf' : (req.file.mime || 'application/pdf');
    if (!/^(application\/pdf|image\/(png|jpe?g|webp))$/.test(mime)) throw new Error('Upload a PDF or a photo (JPG/PNG) of your resume.');
    parts.push({ inlineData: { mimeType: mime, data: req.file.data } });
    try { fileUrl = folder_('Uploads').createFile(Utilities.newBlob(bytes, mime, session + ' - ' + (req.file.name || 'resume'))).getUrl(); } catch (e) { fileUrl = ''; }
  }
  contents.push({ role: 'user', parts: parts });

  const recent = (req.history || []).slice(-2).map(h => h.text).join(' ');
  const out = gemini_(systemPrompt_(name, msg + ' ' + recent), contents,
    mode === 'interview' ? { temperature: 0.5, maxOutputTokens: 1200, thinkingConfig: { thinkingBudget: 0 } } : null);
  if (BUSY_RE.test(out)) return { reply: out, flag: 'busy' };
  const flag = /\[HANDOFF\]/.test(out) ? 'handoff' : /\[CRISIS\]/.test(out) ? 'crisis' : '';
  const reply = out.replace(/\[(HANDOFF|CRISIS)\]/g, '').trim();
  const logMsg = mode === 'interview' ? msg.replace(/^\[[\s\S]*?\]\s*/, '').replace(/^MY ANSWER:\s*/, '') : msg;
  tab_(ss_(), 'chats').appendRow([new Date(), session, name, mode, logMsg.slice(0, 1000) + (req.file ? ' [file: ' + (req.file.name || '') + ']' : ''), reply.slice(0, 4000), flag, fileUrl]);
  if (/^\s*\**\s*REPORT/i.test(reply)) { try {
    const sc = (reply.match(/SCORE:\s*(\d+(?:\.\d)?)/i) || [])[1] || '';
    tab_(ss_(), 'reports').appendRow([new Date(), session, name, String(req.role || ''), String(req.lang || ''), sc, reply.slice(0, 5000)]);
  } catch (e) {} }
  return { reply: reply, flag: flag };
}

function gemini_(system, contents, cfg) {
  const key = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!key) throw new Error('Ila is not set up yet (missing GEMINI_API_KEY).');
  const models = [CONFIG.MODEL].concat(CONFIG.BACKUP_MODELS || []);
  const gc = cfg || { temperature: 0.6, maxOutputTokens: 700, thinkingConfig: { thinkingBudget: 0 } };  // no "thinking" = fast chat replies
  let body = JSON.stringify({ systemInstruction: { parts: [{ text: system }] }, contents: contents, generationConfig: gc });
  let last = 0;
  for (let m = 0; m < models.length; m++) {
    for (let t = 0; t < 2; t++) {
      const resp = UrlFetchApp.fetch('https://generativelanguage.googleapis.com/v1beta/models/' +
        models[m] + ':generateContent', { method: 'post', contentType: 'application/json',
        headers: { 'x-goog-api-key': key }, muteHttpExceptions: true, payload: body });
      last = resp.getResponseCode();
      if (last === 200) {
        const c = (JSON.parse(resp.getContentText()).candidates || [])[0];
        return c && c.content ? c.content.parts.map(p => p.text || '').join('')
          : 'Sorry, I could not answer that. Could you rephrase?';
      }
      if (last === 400 && gc.thinkingConfig) {   // this model doesn't accept the speed setting: drop it and retry
        delete gc.thinkingConfig;
        body = JSON.stringify({ systemInstruction: { parts: [{ text: system }] }, contents: contents, generationConfig: gc });
        t--; continue;
      }
      if (last === 400 || last === 403) throw new Error('Ila could not answer (' + last + '). Check the API key.');
      if (last === 404) break;                 // model name not available, try the next one
      Utilities.sleep(800);         // 429 / 500 / 503: wait, then retry
    }
  }
  return 'Ila is a little busy right now (Google is overloaded). Please try again in a minute.';
}

function ss_() {
  const id = String(PropertiesService.getScriptProperties().getProperty('SHEET_ID') || CONFIG.SHEET_ID || '').trim();
  return id ? SpreadsheetApp.openById(id) : SpreadsheetApp.getActiveSpreadsheet();
}
function tab_(ss, k) {
  const [name, headers] = TABS[k];
  let s = ss.getSheetByName(name);
  if (!s) {
    s = ss.insertSheet(name);
    s.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold').setBackground('#0B3D1E').setFontColor('#FFFFFF');
    s.setFrozenRows(1);
  } else if (s.getLastColumn() < headers.length) {
    s.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold').setBackground('#0B3D1E').setFontColor('#FFFFFF');
  }
  return s;
}
function overLimit_(session) {
  const c = CacheService.getScriptCache(), key = 'n_' + session + '_' + Utilities.formatDate(new Date(), 'Asia/Kolkata', 'yyyyMMdd');
  const n = Number(c.get(key) || 0) + 1;
  c.put(key, String(n), 21600);
  return n > CONFIG.DAILY_MESSAGE_LIMIT;
}

function anDays_(rows, dateCol, n) {
  var out = [], idx = {}, tz = 'Asia/Kolkata', now = new Date();
  for (var i = n - 1; i >= 0; i--) { var d = new Date(now.getTime() - i * 864e5), k = Utilities.formatDate(d, tz, 'yyyy-MM-dd');
    idx[k] = out.length; out.push({ k: k, label: Utilities.formatDate(d, tz, 'd MMM'), rows: [] }); }
  rows.forEach(function (r) { try { var k = Utilities.formatDate(new Date(r[dateCol]), tz, 'yyyy-MM-dd'); if (k in idx) out[idx[k]].rows.push(r); } catch (e) {} });
  return out;
}
function anEsc_(x) { return String(x == null ? '' : x).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
/** Bar chart; optional line (second series, own scale 0..lineMax). */
function anChart_(labels, bars, line, lineMax, barName, lineName) {
  var W = 640, H = 200, P = 28, n = labels.length, bw = (W - P * 2) / n, max = Math.max.apply(null, bars.concat([1]));
  var g = '<svg viewBox="0 0 ' + W + ' ' + (H + 36) + '" width="100%" role="img" style="display:block">';
  [0, .5, 1].forEach(function (f) { var y = P + (H - P * 2) * (1 - f); g += '<line x1="' + P + '" x2="' + (W - P) + '" y1="' + y + '" y2="' + y + '" stroke="#DCE5DE"/>' +
    '<text x="' + (P - 6) + '" y="' + (y + 4) + '" font-size="10" text-anchor="end" fill="#8A968E">' + Math.round(max * f) + '</text>'; });
  bars.forEach(function (v, i) { var h = (H - P * 2) * v / max, x = P + i * bw + bw * .18;
    g += '<rect x="' + x + '" y="' + (H - P - h) + '" width="' + (bw * .64) + '" height="' + h + '" rx="3" fill="#16A34A"><title>' + anEsc_(labels[i]) + ': ' + v + '</title></rect>'; });
  if (line) { var pts = []; line.forEach(function (v, i) { if (v == null) return; pts.push((P + i * bw + bw / 2) + ',' + (P + (H - P * 2) * (1 - v / lineMax))); });
    if (pts.length) g += '<polyline points="' + pts.join(' ') + '" fill="none" stroke="#0B3D1E" stroke-width="2.5"/>' + pts.map(function (p) { var a = p.split(','); return '<circle cx="' + a[0] + '" cy="' + a[1] + '" r="3.5" fill="#0B3D1E"/>'; }).join(''); }
  var step = Math.ceil(n / 8);
  labels.forEach(function (l, i) { if (i % step === 0 || (i === n - 1 && (n - 1) % step > step / 2)) g += '<text x="' + (P + i * bw + bw / 2) + '" y="' + (H - P + 16) + '" font-size="10" text-anchor="middle" fill="#8A968E">' + anEsc_(l) + '</text>'; });
  g += '<rect x="' + P + '" y="' + (H + 12) + '" width="10" height="10" fill="#16A34A"/><text x="' + (P + 14) + '" y="' + (H + 21) + '" font-size="11" fill="#2C3A31">' + anEsc_(barName) + '</text>';
  if (line) g += '<line x1="' + (P + 150) + '" x2="' + (P + 164) + '" y1="' + (H + 17) + '" y2="' + (H + 17) + '" stroke="#0B3D1E" stroke-width="2.5"/><text x="' + (P + 168) + '" y="' + (H + 21) + '" font-size="11" fill="#2C3A31">' + anEsc_(lineName) + '</text>';
  return g + '</svg>';
}
/** Horizontal bars for a {label: count} map. */
function anHBars_(map, top) {
  var arr = Object.keys(map).map(function (k) { return [k, map[k]]; }).sort(function (a, b) { return b[1] - a[1]; }).slice(0, top || 6);
  var max = arr.length ? arr[0][1] : 1;
  return arr.length ? arr.map(function (a) { return '<div style="display:flex;align-items:center;gap:10px;margin:6px 0;font-size:13px"><span style="width:44%;color:#0B3D1E">' + anEsc_(a[0]) +
    '</span><div style="flex:1;height:10px;background:#DCE5DE;border-radius:6px;overflow:hidden"><i style="display:block;height:100%;width:' + Math.round(a[1] / max * 100) + '%;background:#16A34A"></i></div><b style="width:30px;text-align:right">' + a[1] + '</b></div>'; }).join('')
    : '<p style="color:#8A968E;font-size:13px">No data yet.</p>';
}

}
